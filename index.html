// ------------------------------------------------------------------    // Hexagon Coordinate Helper (Axial -> World)    // ------------------------------------------------------------------    function axialToWorld(q, r) {      const x = Math.sqrt(3) * SPACING * (q + r / 2);      const z = 1.5 * SPACING * r;      return { x, z };    }

// Procedural Ground Canvas Texture    function createGroundCanvasTexture(type) {      const canvas = document.createElement('canvas');      canvas.width = 256; canvas.height = 256;      const ctx = canvas.getContext('2d');      ctx.fillStyle = "#e8dac8";      ctx.fillRect(0, 0, 256, 256);

if (type === 0) { // Brick Pattern        ctx.strokeStyle = "rgba(180, 150, 120, 0.25)";        ctx.lineWidth = 4;        for (let i = 0; i < 256; i += 32) {          ctx.beginPath(); ctx.moveTo(0, i); ctx.lineTo(256, i); ctx.stroke();          for (let j = 0; j < 256; j += 32) {            ctx.beginPath(); ctx.moveTo(j + (i % 64 ? 16 : 0), i); ctx.lineTo(j + (i % 64 ? 16 : 0), i + 32); ctx.stroke();          }        }      } else if (type === 1) { // Dots        ctx.fillStyle = "rgba(160, 130, 100, 0.15)";        for (let x = 16; x < 256; x += 32) {          for (let y = 16; y < 256; y += 32) {            ctx.beginPath(); ctx.arc(x, y, 5, 0, Math.PI * 2); ctx.fill();          }        }      } else { // Stripes        ctx.strokeStyle = "rgba(170, 140, 110, 0.2)";        ctx.lineWidth = 6;        for (let i = -256; i < 512; i += 24) {          ctx.beginPath(); ctx.moveTo(i, 0); ctx.lineTo(i + 256, 256); ctx.stroke();        }      }      const tex = new THREE.CanvasTexture(canvas);      tex.wrapS = THREE.RepeatWrapping;      tex.wrapT = THREE.RepeatWrapping;      return tex;    }

// ------------------------------------------------------------------    // Sky Shader, Lighting & Atmosphere    // ------------------------------------------------------------------    function buildSkyAndLighting() {      // Hemisphere & Directional Sun Light      hemiLight = new THREE.HemisphereLight(0xeef6fa, 0x8f8270, 1.3);      scene.add(hemiLight);

sunLight = new THREE.DirectionalLight(0xfff1de, 2.8);      sunLight.castShadow = true;      sunLight.shadow.mapSize.width = 2048;      sunLight.shadow.mapSize.height = 2048;      sunLight.shadow.camera.near = 1;      sunLight.shadow.camera.far = 90;      sunLight.shadow.camera.left = -20;      sunLight.shadow.camera.right = 20;      sunLight.shadow.camera.top = 22;      sunLight.shadow.camera.bottom = -22;      sunLight.shadow.bias = -0.0006;      sunLight.shadow.normalBias = 0.045;      sunLight.shadow.radius = 3;      scene.add(sunLight);

const fillLight = new THREE.DirectionalLight(0xbfd6ff, 0.35);      fillLight.position.set(-20, 15, -20);      scene.add(fillLight);

// Sky Atmosphere Sphere Shader      const skyGeo = new THREE.SphereGeometry(170, 32, 15);      skyUniforms = {        uTop: { value: new THREE.Color() },        uHorizon: { value: new THREE.Color() },        uSunDir: { value: new THREE.Vector3() },        uNight: { value: 0 }      };

const skyMat = new THREE.ShaderMaterial({        uniforms: skyUniforms,        side: THREE.BackSide,        depthWrite: false,        vertexShader: `          varying vec3 vWorldPosition;          void main() {            vec4 worldPosition = modelMatrix * vec4(position, 1.0);            vWorldPosition = worldPosition.xyz;            gl_Position = projectionMatrix * viewMatrix * worldPosition;          }        `,        fragmentShader: `          uniform vec3 uTop;          uniform vec3 uHorizon;          uniform vec3 uSunDir;          uniform float uNight;          varying vec3 vWorldPosition;

float hash(vec3 p) {            p  = fract(p * 0.3183099 + .1);            p *= 17.0;            return fract(p.x * p.y * p.z * (p.x + p.y + p.z));          }

void main() {            vec3 dir = normalize(vWorldPosition);            float h = smoothstep(-0.02, 0.55, dir.y);            vec3 col = mix(uHorizon, uTop, h);

// Sun Glow            float sunDot = max(0.0, dot(dir, uSunDir));            col += vec3(1.0, 0.8, 0.5) * pow(sunDot, 18.0) * 0.35;

// Night Stars            if (uNight > 0.1) {              vec3 starP = floor(dir * 180.0);              if (hash(starP) > 0.9965 && dir.y > 0.1) {                col += vec3(0.9, 0.95, 1.0) * uNight * (0.5 + 0.5 * sin(dir.x * 100.0));              }            }            gl_FragColor = vec4(col, 1.0);          }        `      });      skyMesh = new THREE.Mesh(skyGeo, skyMat);      scene.add(skyMesh);    }

// Dynamic Shader Water Plane    function buildWater() {      const waterGeo = new THREE.CircleGeometry(160, 64);      waterGeo.rotateX(-Math.PI / 2);

waterUniforms = {        uTime: { value: 0 },        uWaterColor: { value: new THREE.Color("#a4d4da") },        fogColor: { value: scene.fog.color },        fogNear: { value: scene.fog.near },        fogFar: { value: scene.fog.far }      };

const waterMat = new THREE.ShaderMaterial({        uniforms: waterUniforms,        vertexShader: `          varying vec3 vWorldPos;          void main() {            vec4 wp = modelMatrix * vec4(position, 1.0);            vWorldPos = wp.xyz;            gl_Position = projectionMatrix * viewMatrix * wp;          }        `,        fragmentShader: `          uniform float uTime;          uniform vec3 uWaterColor;          uniform vec3 fogColor;          uniform float fogNear;          uniform float fogFar;          varying vec3 vWorldPos;

void main() {            float wave = sin(vWorldPos.x * 0.4 + uTime * 1.2) * cos(vWorldPos.z * 0.4 + uTime * 1.5) * 0.5 + 0.5;            vec3 col = mix(uWaterColor, uWaterColor * 0.8, wave);                        // Distance fog blend            float dist = length(vWorldPos.xz);            float fogFactor = smoothstep(fogNear, fogFar, dist);            col = mix(col, fogColor, fogFactor);

gl_FragColor = vec4(col, 1.0);          }        `      });

waterMesh = new THREE.Mesh(waterGeo, waterMat);      waterMesh.position.y = WATER_Y;      waterMesh.receiveShadow = true;      scene.add(waterMesh);    }

// Background Basin Terrain & Distant Mountains    function buildTerrainAndDistantMountains() {      const terrainGeo = new THREE.PlaneGeometry(170, 170, 120, 120);      terrainGeo.rotateX(-Math.PI / 2);      const pos = terrainGeo.attributes.position;      const colors = [];

const cRiver = new THREE.Color("#8fb3ad");      const cSand  = new THREE.Color("#d9c9a3");      const cFarm  = new THREE.Color("#aec493");      const cGrass = new THREE.Color("#a3b986");      const cForest= new THREE.Color("#6f8a62");      const cRock  = new THREE.Color("#b2a38d");

for (let i = 0; i < pos.count; i++) {        const x = pos.getX(i);        const z = pos.getZ(i);        const dist = Math.sqrt(x * x + z * z);                // Ring Mountains Ring Formation        let h = 0;        if (dist > 25) {          const ringFactor = THREE.MathUtils.smoothstep(dist, 25, 55);          const n = simplex.noise2D(x * 0.03, z * 0.03);          h = ringFactor * (12 + n * 18);          if (z > 20) h *= 0.3; // Flatten front camera view        }

pos.setY(i, h);

// Vertex Color Grading based on elevation        let col = cGrass;        if (h < 0.2) col = cRiver;        else if (h < 1.5) col = cSand;        else if (h < 4.0) col = cFarm;        else if (h < 12.0) col = cForest;        else col = cRock;

colors.push(col.r, col.g, col.b);      }

terrainGeo.computeVertexNormals();      terrainGeo.setAttribute('color', new THREE.Float32BufferAttribute(colors, 3));

const terrainMat = new THREE.MeshStandardMaterial({        vertexColors: true,        roughness: 0.9,        flatShading: true      });

const terrainMesh = new THREE.Mesh(terrainGeo, terrainMat);      terrainMesh.position.y = -0.5;      scene.add(terrainMesh);    }

const deckGeo = new THREE.BoxGeometry(0.5, 0.08, dist * 0.52);      const deckMat = new THREE.MeshStandardMaterial({ color: 0xb48560, roughness: 0.8 });      const deckMesh = new THREE.Mesh(deckGeo, deckMat);      deckMesh.position.y = 0.02;      deckMesh.castShadow = true;      bridgeGroup.add(deckMesh);

scene.add(bridgeGroup);    }

// ------------------------------------------------------------------    // 6. Chibi Walkers & Flow Connection Tubes    // ------------------------------------------------------------------    function buildWalkersAndFlows() {      DATA.flows.forEach(flow => {        const isFrom = islandMap.get(flow.from);        const isTo = islandMap.get(flow.to);        if (!isFrom || !isTo) return;

// Tube Curve Path        const pA = isFrom.worldPos.clone().add(new THREE.Vector3(0, 0.1, 0));        const pB = isTo.worldPos.clone().add(new THREE.Vector3(0, 0.1, 0));        const mid = pA.clone().add(pB).multiplyScalar(0.5).add(new THREE.Vector3(0, 0.8, 0));

const curve = new THREE.QuadraticBezierCurve3(pA, mid, pB);        const tubeGeo = new THREE.TubeGeometry(curve, 20, 0.035, 8, false);

const flowMat = new THREE.ShaderMaterial({          transparent: true,          uniforms: {            uTime: { value: 0 },            uColor: { value: new THREE.Color("#db6a41") },            uStyle: { value: flow.evidence === 'official' ? 0 : (flow.evidence === 'thematic' ? 1 : 2) }          },          vertexShader: `            varying vec2 vUv;            void main() {              vUv = uv;              gl_Position = projectionMatrix * viewMatrix * modelMatrix * vec4(position, 1.0);            }          `,          fragmentShader: `            uniform float uTime;            uniform vec3 uColor;            uniform int uStyle;            varying vec2 vUv;

void main() {              float u = vUv.x * 25.0 - uTime * 3.0;              float alpha = 0.82;

if (uStyle == 1 && fract(u) > 0.6) discard; // Dashed              if (uStyle == 2 && fract(u) > 0.3) discard; // Dotted

gl_FragColor = vec4(uColor, alpha);            }          `        });

const tubeMesh = new THREE.Mesh(tubeGeo, flowMat);        tubeMesh.visible = false; // Hidden by default        scene.add(tubeMesh);        flowTubes.push({ data: flow, mesh: tubeMesh, mat: flowMat });

// Add Chibi Walker Character        createChibiWalker(curve, flow);      });    }

// ------------------------------------------------------------------    // Day/Night Keyframe Interpolation & Sky Update    // ------------------------------------------------------------------    function updateAtmosphere(mins) {      let k1 = DAY_KEYFRAMES[0], k2 = DAY_KEYFRAMES[1];      for (let i = 0; i < DAY_KEYFRAMES.length - 1; i++) {        if (mins >= DAY_KEYFRAMES[i].min && mins <= DAY_KEYFRAMES[i + 1].min) {          k1 = DAY_KEYFRAMES[i];          k2 = DAY_KEYFRAMES[i + 1];          break;        }      }

const factor = (mins - k1.min) / (k2.min - k1.min || 1);

// Lerp Sky Colors & Lights      skyUniforms.uTop.value.setStyle(k1.top).lerp(new THREE.Color(k2.top), factor);      skyUniforms.uHorizon.value.setStyle(k1.horiz).lerp(new THREE.Color(k2.horiz), factor);            const sunC = new THREE.Color(k1.sunC).lerp(new THREE.Color(k2.sunC), factor);      sunLight.color.copy(sunC);      sunLight.intensity = THREE.MathUtils.lerp(k1.sunI, k2.sunI, factor);

hemiLight.color.setStyle(k1.hemiS).lerp(new THREE.Color(k2.hemiS), factor);      hemiLight.groundColor.setStyle(k1.hemiG).lerp(new THREE.Color(k2.hemiG), factor);      hemiLight.intensity = THREE.MathUtils.lerp(k1.hemiI, k2.hemiI, factor);

const nightVal = THREE.MathUtils.lerp(k1.night, k2.night, factor);      skyUniforms.uNight.value = nightVal;

waterUniforms.uWaterColor.value.setStyle(k1.water).lerp(new THREE.Color(k2.water), factor);      scene.fog.color.copy(skyUniforms.uHorizon.value);

// Sun Orbit Math      const t = Math.max(0, Math.min(1, (mins - 360) / 720));      const az = Math.PI * (0.15 + 0.7 * t);      const el = 0.35 + Math.sin(t * Math.PI) * 0.75;      const dir = new THREE.Vector3(Math.cos(az) * Math.cos(el), Math.sin(el), 0.55 * Math.cos(az) + 0.35).normalize();

sunLight.position.copy(dir).multiplyScalar(40);      skyUniforms.uSunDir.value.copy(dir);    }

// ------------------------------------------------------------------    // 9. Main Animation Loop    // ------------------------------------------------------------------    function animate(time) {      const dt = 0.016;

// 1. Advance Time (144s per day cycle)      currentTimeMinutes = (currentTimeMinutes + dt * 10) % 1440;      updateAtmosphere(currentTimeMinutes);

// 2. Camera Flying Interpolation      if (isAnimatingCam) {        const elapsed = performance.now() - camAnimStartTime;        const progress = Math.min(elapsed / 800, 1);        const easeProgress = progress < 0.5 ? 4 * progress * progress * progress : 1 - Math.pow(-2 * progress + 2, 3) / 2;

camera.position.lerpVectors(camStartPos, camTargetPos, easeProgress);        controls.target.lerpVectors(camStartLook, camTargetLook, easeProgress);

if (progress >= 1) isAnimatingCam = false;      }

controls.update();

// 3. Update Dynamic Walkers      walkers.forEach(w => {        w.progress = (w.progress + dt * w.speed * 0.1) % 1;        const pt = w.path.getPoint(w.progress);        w.group.position.copy(pt);      });

// 4. Flow Shader Animation      flowTubes.forEach(ft => {        ft.mat.uniforms.uTime.value = time * 0.001;      });

waterUniforms.uTime.value = time * 0.001;

// Render Scene      renderer.render(scene, camera);      cssRenderer.render(scene, camera);    }
